% --- Code-path typesetting --------------------------------------------------
% \codepath breaks long monospace identifiers (file paths, function names,
% config keys) at any character so unbreakable underscores and slashes do
% not cause overfull-hbox bleeds in narrow appendix columns and prose. The
% sloppy + emergencystretch combination is applied at the paragraph level
% via an \appendixoverflowsafe declaration loaded inside each appendix.
\usepackage{seqsplit}
\newcommand{\codepath}[1]{\texttt{\seqsplit{#1}}}
% Apply once per appendix paragraph that contains long monospace identifiers
% to license aggressive whitespace stretching as a last-resort fallback.
\newcommand{\appendixoverflowsafe}{%
  \sloppy
  \emergencystretch=5em
  \hbadness=10000
}
