\section*{Problem and gap}

\begin{frame}{The problem}
  \begin{itemize}\small
    \item Large language models hallucinate at \textbf{3\%--91\%} across deployment domains
          \tiny\cite{alkaissi2023artificial,10.1145/3649506}\normalsize.
    \item TruthfulQA shows hallucination is \textbf{not solved by scale} -- bigger models often hallucinate \emph{more confidently} \tiny\cite{lin-etal-2022-truthfulqa}\normalsize.
    \item Existing fixes are \alert{post-hoc detectors}: they label outputs but cannot \emph{change what the system serves}.
  \end{itemize}

  \vspace{0.5em}
  \begin{block}{Thesis claim}
    Treat hallucination as a \textbf{system-level architectural} problem, not a detection problem.
    Pair the model with an \emph{episodic memory}, a \emph{calibrated nine-signal verifier with per-benchmark composite}, a \emph{fixed-threshold storage gate on calibrated probability}, and a \emph{cycle-boundary self-improvement loop} (LoRA-based) that learns from its own verified outputs.
  \end{block}
\end{frame}

% =====================================================================
\begin{frame}{The three structural shortcomings of stateless inference}
  Standard inference suffers \alert{three reinforcing shortcomings} (Ch1 \S stateless-gap):

  \vspace{0.5em}
  \begin{enumerate}\small
    \item \textbf{Statelessness.} Every query processed independently. \emph{No representation of past successful answers retained}. The system wastes computation on questions it has effectively already solved, and identical questions can produce inconsistent answers from one session to the next.
    \item \textbf{No calibrated confidence.} The model emits a string without a per-query reliability signal. The user cannot tell trustworthy outputs apart from confident-but-wrong ones.
    \item \textbf{No self-improvement mechanism.} External verifications (whether by humans or other systems) decay into one-shot interventions. The model learns nothing from its own history of right and wrong answers.
  \end{enumerate}

  \vspace{0.5em}
  \begin{block}{What CAEM gives back}
    Memory closes (1). Calibrated $u_{\text{stored}}$ + four-outcome decision closes (2). Cycle-boundary SIL fine-tune closes (3). Each axis reinforces the others -- without memory the model cannot reuse verified knowledge; without calibration the system cannot recognise when memory is missing; without self-improvement, knowledge accumulated stays static.
  \end{block}
\end{frame}

% =====================================================================
\begin{frame}{CAEM at a glance -- two interlocking loops}
  \begin{columns}[t]
    \column{0.55\textwidth}
    \textbf{Per-query path (8 stages, $\sim$10\,s):}
    \begin{enumerate}\setlength\itemsep{0pt}\scriptsize
      \item Pre-routing confidence $u_{\text{pre}}$ (per-bench $T_b$)
      \item Encode + episodic-memory lookup
      \item Three-tier router (with safety override)
      \item Tier execution (memory / zero-shot / RAG)
      \item Nine-signal verifier ensemble (per-bench dispatch)
      \item Per-bench composite + fixed-threshold gate $\to$ decision
      \item Output (STORE / DEFER / DISCARD / ABSTAIN)
      \item Per-query housekeeping (memory write)
    \end{enumerate}

    \column{0.45\textwidth}
    \textbf{Per-cycle update (every 4700 queries):}
    \begin{itemize}\setlength\itemsep{0pt}\scriptsize
      \item SIL fine-tune (LoRA r=32, $\alpha$=64, no L$_2$ anchor) + multi-modal retention guard
      \item Cal-fold scoring under post-SIL model
      \item Re-fit per-bench $T_b$ + per-bench isotonic curves (Cherian boost re-fit, shrinkage prior $\alpha{=}0.6$ toward pooled)
      \item Hot-reload verifier
      \item Retroactive re-verify all stored entries
      \item Stream chunk (4700: FEVER 2000 + TQA 2000 + CSQA 700) + held-out eval (1500: 300 $\times$ 5 benches)
    \end{itemize}
  \end{columns}

  \vspace{0.4em}
  {\scriptsize Backbone: Qwen-2.5-3B-Instruct. Verifier judges: MiniCheck (Flan-T5-Large) on short premises; frozen Qwen NLI on longer ones, length-dispatched. Encoder: SBERT all-mpnet-base-v2. Retriever: 21M-passage DPR over Wikipedia. Reranker: BAAI/bge-reranker-v2-m3.}
\end{frame}

\section*{Goals, requirements, hypotheses, commitments}

\begin{frame}{Specific objectives 1--4 (Ch1 \S specific-objectives)}
  \scriptsize
  \begin{enumerate}
    \item \textbf{Verified episodic memory with a four-outcome decision.} Admission rule finer than a binary threshold; every generated answer enters one of \{STORE, DEFERRED, DISCARD, ABSTAIN\} via the calibrated probability composite + the confabulation early-exit. Storage is similarity-indexed with novelty filter and value-based pruning combining recency, retrieval frequency, and stored-answer quality.
    \item \textbf{Pre-generation confidence and three-tier routing with safety override.} $u_{\text{pre}}$ from a single decoder-only forward pass, temperature-scaled per benchmark on a disjoint fold, used \emph{independently} from memory similarity. Router combines query-to-memory similarity with stored-answer quality into a single dispatch score over \{memory-direct, zero-shot, RAG\}. Independent safety override forces RAG whenever $u_{\text{pre}}<u_{\text{pre,min}}^b$ -- the per-benchmark fitted floor (currently $0.38$ on every training bench) -- grounded retrieval is available before generation begins.
    \item \textbf{Nine-signal post-generation verifier with per-benchmark calibrated probability composite.} Four families: internal calibration ($u_{\text{token}}, u_{\text{dropout}}, u_{\text{internal}}$), sample-set agreement ($s_{\text{avg}}, p_{\text{entail}}$), external grounding ($p_{\text{ground,max}}, p_{\text{ground,mean}}, p_{\text{ground,atomic}}$), and question-answer relevance ($q_{\text{a,rel}}$). Each training benchmark has its own per-signal isotonic curves and Cherian L$_2$ logistic boost weights/intercept, shrunk toward a pooled prior at $\alpha{=}0.6$. Transfer benchmarks use the pooled global composite.
    \item \textbf{Fixed-threshold storage gate on the calibrated probability.} The composite output decides a four-way outcome: $u_{\text{stored}} \ge 0.60 \Rightarrow$ STORE, $\ge 0.45 \Rightarrow$ DEFERRED, otherwise ABSTAIN or DISCARD (split by a passage-grounding floor). The threshold is constant across cycles; the per-bench signal-to-probability mapping refits at each cycle boundary so that the same threshold means the same precision target as the model improves.
  \end{enumerate}
\end{frame}

% =====================================================================
\begin{frame}{Specific objectives 5--7 (Ch1 \S specific-objectives)}
  \scriptsize
  \begin{enumerate}\setcounter{enumi}{4}
    \item \textbf{Retroactive review and deferred reconsideration.} Verifier internal-calibration signals read the generator's hidden state, so stored composites drift after every fine-tune. At every cycle boundary, retroactive re-verification rescores every stored entry under the freshly recalibrated verifier and prunes entries below $\tau_{\text{retro}}=0.50$. A deferred-buffer pass promotes entries that have now crossed $\tau_{\text{store}}$ and discards entries past their time-to-live. The orchestrator hard-fails if the deferred pass does not fire, so the mechanism cannot silently lapse.
    \item \textbf{Constrained self-improvement with multi-modal retention guard.} The training pool draws from stored episodes whose calibrated probability exceeds $\tau_{\text{train}}=0.75$, strictly above the storage threshold so that borderline entries cannot contaminate the gradient signal. Fine-tune is LoRA-only: rank 32, alpha 64, all-linear targets. The LoRA parameter budget acts as the implicit anchor, so no explicit L$_2$ term is needed. A held-out multi-modal retention probe (MMLU, TriviaQA-test, CommonsenseQA-test) runs before and after; if any probe drops below $\rho_{\min}=0.93$ of pristine, the cycle aborts and the adapter is rolled back. Memory survives rollback.
    \item \textbf{Empirical validation against external baselines and ablations.} Five-benchmark active panel: three training (FEVER, TriviaQA, CommonsenseQA) and two transfer (TruthfulQA, StrategyQA). Multi-modal retention probe spans bounded-label and open-text generation. Composite hallucination metric, per-tier behaviour, retention trajectory, and verifier-accuracy trajectory reported cycle by cycle. Seven external baselines (B1 through B7) evaluated under matched protocol, with paired McNemar plus BCa bootstrap under Holm correction, alongside the pre-registered architectural ablations.
  \end{enumerate}

  \vspace{0.3em}
  {\scriptsize Each objective is a separately-checkable empirical deliverable. The empirical chapter reports the receipt for every one.}
\end{frame}

\begin{frame}{Six design commitments (Ch3 \S constraints)}
  \scriptsize
  \begin{enumerate}
    \item \textbf{Single-backbone commitment.} Generator FIXED across every cycle and every tier. \emph{All reported gains are attributable to memory + calibration + the loop, not to a backbone swap.}
    \item \textbf{Per-benchmark calibration with shrinkage prior commitment.} The signal-to-probability composite is fitted per benchmark on a labelled calibration fold and shrunk at $\alpha{=}0.6$ toward a pooled prior. Storage thresholds (0.60/0.45) and training-pool threshold (0.75) are identical across benchmarks. \emph{No per-benchmark threshold leakage; per-benchmark specialisation lives at the signal-to-probability mapping layer only.}
    \item \textbf{Local-execution commitment.} No external model API in the inference path. Verifier ensemble + embedding encoder + passage index + generator all run from open weights on a single rented GPU. \emph{Deployable without third-party dependencies.}
    \item \textbf{Bounded-cycle commitment.} SIL loop capped at 10 cycles; early stop only on the registered equilibrium diagnostic. \emph{Ensures the empirical claim is not propped up by an arbitrary horizon.}
    \item \textbf{Matched-protocol commitment.} Every external baseline evaluated under same samples, prompts, scoring rules. \emph{The comparison panel is not confounded by re-implementation choices.}
    \item \textbf{Single-seed reproducibility.} Random-state surface capped at one recorded seed; per-cycle artefact catalogue + pre-Step-7 snapshot on a public dataset host. \emph{Trajectory regenerable end-to-end by a third party.}
  \end{enumerate}
\end{frame}

\section*{Per-stage deep dive}

\begin{frame}{Stage 1 -- pre-routing confidence $u_{\text{pre}}$}
  \footnotesize
  Two cheap signals from one short forward pass:
  \begin{itemize}\small
    \item $u_{\text{token}}$ -- geometric mean of per-token log-probabilities of the draft answer
    \item $c_{\text{conv}}$ -- encoder-layer convergence ratio (variance early over variance late layers)
  \end{itemize}
  \[
    u_{\text{pre}} = \sigma\!\left(\frac{\operatorname{logit}\bigl(0.60\,u_{\text{token}} + 0.40\,(1{+}c_{\text{conv}})^{-1}\bigr)}{T_b}\right)
  \]
  Per-benchmark temperature $T_b$ refit every cycle by ECE-minimisation on the calibration fold. A per-benchmark safety floor $u_{\text{pre,min}}^b$ is also fitted from that fold and feeds the router's safety override.

  \vspace{0.2em}
  \why{Per-benchmark $T_b$: FEVER's three-way label space, TriviaQA's bare-entity surface, and CommonsenseQA's five-choice MCQ produce different token-confidence distributions; a single pooled temperature mis-scales the open-text benchmark.}

  \altn{A single-signal $u_{\text{pre}} = u_{\text{token}}$ misses the encoder-layer instability that confidently-wrong outputs exhibit. Adding $c_{\text{conv}}$ catches a class token probability alone cannot.}
\end{frame}

% =====================================================================
\begin{frame}{Stages 2--3 -- memory and the three-tier router}
  \footnotesize
  \textbf{Memory:} FAISS index over SBERT embeddings of past verified $(q,a)$ pairs. Top-1 returns similarity $s$ and neighbour's $u_{\text{stored,nbr}}$.

  \vspace{0.2em}
  \textbf{Combined dispatch score:} $\mathcal{S} = 0.70\,s + 0.30\,u_{\text{stored,nbr}}$.

  \vspace{0.2em}
  \textbf{Dispatch rules in fixed order:}
  \begin{enumerate}\small
    \item Safety override: if $u_{\text{pre}} < u_{\text{pre,min}}^b$ (per-benchmark floor), force Tier 3 unconditionally.
    \item Tier 1 (memory-direct) if $\mathcal{S} \ge 0.90$ -- serve the neighbour answer, no generation, no verifier.
    \item Tier 2 (zero-shot) if similarity $> 0.75$ and not Tier 1 -- generate without retrieval.
    \item Tier 3 (RAG) -- fallback floor.
  \end{enumerate}

  \why{Three tiers because a binary memory-versus-generate decision over-burdens a single threshold. Tier 2 absorbs the ``in-distribution but novel question'' band where the model knows zero-shot but no neighbour is close enough.}

  \why{Safety override is OR, not AND: model readiness and memory quality are both mandatory. Low confidence wins regardless of how good the neighbour looks. Each benchmark sets its own minimum confidence floor from its observed signal distribution.}
\end{frame}

% =====================================================================
\begin{frame}{Stage 4 -- tier execution: why three handlers}
  \scriptsize
  \begin{tabular}{@{}p{1.7cm}p{4.2cm}p{6.5cm}@{}}
    \toprule
    \textbf{Tier} & \textbf{What it does} & \textbf{Cost / when chosen} \\
    \midrule
    Tier 1 (memory) & Returns neighbour's stored answer; verifier skipped & $\sim$50 ms / query;
                    chosen when memory has match committed under $u_{\text{stored}}\ge\tau_{\text{store}}$ (combined $\ge 0.90$). \\
    Tier 2 (zero-shot) & Single forward pass, no retrieval; verifier runs & $\sim$3--5 s / query;
                    in-distribution queries where the model's own knowledge suffices. \\
    Tier 3 (RAG) & Retrieve from 21M passages, rerank, generate grounded; verifier runs & $\sim$10 s / query;
                    architectural floor when nothing else has earned it. \\
    \bottomrule
  \end{tabular}

  \vspace{0.5em}
  \textbf{Cost--quality compounding} (load-bearing for the ``cost reduction'' claim H5):
  \begin{itemize}\scriptsize
    \item Early cycles: Tier 1 share $\approx 0$\%, Tier 3 share $> 90$\% -- pay full RAG cost.
    \item Late cycles: Tier 1 share grows as the verified pool fills.
    \item Mean per-query latency drops $\propto$ Tier-1 share. Plain fine-tuning still pays a full forward pass forever -- this is one of CAEM's load-bearing differentiators.
  \end{itemize}

  \why{Verifier SKIPPED on Tier 1: the entry was committed under the fixed-threshold storage gate (post-cycle-boundary recalibration) at storage time, so re-verifying every recall would (a) double the compute and (b) be circular. The empirical-precision contract acts like a paid receipt; retroverify at the next cycle boundary catches any entry whose recomputed $u_{\text{stored}}$ has decayed below $\tau_{\text{retro}}{=}0.50$.}
\end{frame}

\begin{frame}{Stage 5 -- Family A: internal calibration}
  \scriptsize
  Three signals from the model itself. White-box, zero extra forward passes (re-uses Stage-1 forward).

  \vspace{0.3em}
  \begin{tabular}{@{}lp{4.5cm}p{5cm}@{}}
    \toprule
    Signal & Definition & Purpose \\
    \midrule
    $u_{\text{token}}$ & geometric mean per-token log-prob of draft & calibrated by SFT, baseline confidence \\
    $u_{\text{dropout}}$ & variance over $K=5$ MC-Dropout passes (rate $0.10$) & catches deterministic-decode-stable but dropout-fragile answers \\
    $u_{\text{internal}}$ & $\tfrac{1}{2}u_{\text{token}} + \tfrac{1}{2}(1{-}u_{\text{dropout}})$ & combined internal stability for early-exit \\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \why{$K=5$ is the published Bayesian-dropout balance (Gal 2016): enough to estimate predictive variance, few enough for one extra forward pass equivalent.}

  \why{$u_{\text{internal}}$ feeds the Stage-6 confabulation early-exit. Linear combination keeps signals in $[0,1]$ for downstream isotonic.}

  \altn{$u_{\text{token}}$ alone misses the over-confident ``I'm sure'' failure mode (Kadavath et al.\ 2022). Adding $u_{\text{dropout}}$ catches it.}
\end{frame}

\begin{frame}{Stage 5 -- Family B: sample-set agreement}
  \scriptsize
  Resample $M{=}3$ chains at $T{=}0.7$. Inter-chain agreement and chain-to-answer entailment.

  \vspace{0.3em}
  \begin{tabular}{@{}lp{4.7cm}p{5cm}@{}}
    \toprule
    Signal & Definition & Closes which failure mode \\
    \midrule
    $s_{\text{avg}}$ & mean unique-pair cosine over $M=3$ chains in SBERT space & ``self-consistency'': model produces same answer on re-roll (Wang et al.\ 2023) \\
    $p_{\text{entail}}$ & mean MiniCheck NLI entailment chain $\to$ served answer & ``served answer is well-formed but is NOT what the chains support'' \\

  \end{tabular}

  \vspace{0.4em}
  \why{$M=3$: $M=1$ gives zero variance, $M=10$ adds little discriminative power per Wang et al. $T=0.7$: greedy collapses chains to identical strings (so $s_{\text{avg}}=1$ trivially); $T=1.0$ is too noisy.}

  \altn{Three candidate signals were considered and excluded from the composite after the calibration audit found their fitted weights statistically indistinguishable from zero: normalised semantic entropy at $K{=}10$, a Wikidata alias-overlap signal, and a head-noun consistency check over the $M$ chains. Including them would have cost compute for no discriminative power.}
\end{frame}

\begin{frame}{Stage 5 -- Family C: external grounding}
  \scriptsize
  Three signals over passages retrieved from 21M-passage Wikipedia DPR + reranked by BAAI/bge-reranker-v2-m3.

  \vspace{0.3em}
  \begin{tabular}{@{}lp{6cm}p{3.5cm}@{}}
    \toprule
    Signal & Pair scored under MiniCheck NLI & Catches \\
    \midrule
    $p_{\text{ground,max}}$ & strongest of top-3 reranked passages $\to$ answer & ``one passage clearly supports'' \\
    $p_{\text{ground,mean}}$ & mean over all top-3 $\to$ answer & ``broad support across passages'' \\
    $p_{\text{ground,atomic}}$ & length-gated atomic-fact decomp $\to$ each top-3 (Min et al.\ 2023) & ``answer claims something not in any passage'' \\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \alert{Important architectural detail:} the reranker scores pairs of $(\text{query} + \text{answer},\ \text{passage})$, not $(\text{query},\ \text{passage})$.

  \why{Concatenating the answer to the query forces the reranker to find passages that could substantiate the \emph{specific claim} the model just produced. If we ranked by topical relevance to the question only, top passages might be on-topic but say nothing about the actual claim.}

  \why{This is also why the rerank is NOT cacheable across cycles: served answer changes after every SIL fine-tune, so the rerank pair differs and top-3 changes. The cost is paid per cycle by design.}
\end{frame}

\begin{frame}{Stage 5 -- Family D + composite math}
  \scriptsize
  \textbf{Family D (one signal):} $q_{\text{a,rel}}$ -- independent cross-encoder score of $(\text{question}, \text{served\_answer})$. Verifier's cross-encoder is FROZEN.

  \why{This single signal carries the largest weight ($w = 0.500$). It catches the ``off-topic but grounded'' failure mode -- model confidently answers a related-but-different question, retrieval finds passages supporting the wrong question. Internal/grounding signals miss this.}

  \vspace{0.4em}
  \textbf{Stage 5b -- per-benchmark composite:}
  \begin{enumerate}\scriptsize
    \item Route by the candidate's benchmark token: if a per-benchmark child composite exists for it, predict through that child; otherwise use the pooled global composite.
    \item Per-signal isotonic $\mathrm{iso}_{j,b}$ produces $P_{j,b}\in(0,1)$. Per-benchmark curves are shrunk toward the pooled prior at $\alpha{=}0.6$: $y_{\text{eff}} = \alpha \cdot y_{\text{per-bench}} + (1{-}\alpha) \cdot y_{\text{pooled}}$.
    \item $L_j = \log\bigl(P_{j,b}/(1{-}P_{j,b})\bigr)$.
    \item Per-benchmark L$_2$ logistic boost (Cherian 2024): $\operatorname{logit}(u_{\text{stored}}) = b_b + \sum_j w_{j,b} L_j$.
    \item Sigmoid back: $u_{\text{stored}} = \sigma(\cdot) \in (0,1)$.
  \end{enumerate}

  \why{Per-benchmark specialisation at both layers: a single pooled fit averages FEVER-style entailment dominance, TriviaQA's bare-entity NLI weakness, and CommonsenseQA's option-text grounding into one mapping that mis-serves at least one of them. Per-benchmark fixes the asymmetry; shrinkage prevents per-benchmark overfitting on the small calibration fold.}

  \why{Shrinkage is applied to the isotonic curves only, not to the boost weights. Isotonic is non-parametric and high-capacity (one knot per signal value), so shrinkage is essential; the per-benchmark logistic coefficients are already heavily regularised by the L$_2$ strength $C=0.01$.}

  \why{$C = 0.01$ chosen by a 25-variant pre-registered sweep over $C \in \{0.01,\ 0.10,\ 1.00\}$, locked by best in-distribution precision. Re-passed at every cycle-boundary refit.}
\end{frame}

\begin{frame}{Stage 6 -- four-outcome decision tree}
  \scriptsize
  \textbf{Fixed-threshold gate on the calibrated probability:}
  \begin{itemize}\scriptsize
    \item $\tau_{\text{store}} = 0.60$
    \item $\tau_{\text{defer}} = 0.45$
    \item Thresholds are constant across cycles; per-benchmark refit happens at the signal-to-probability composite, not at the gate.
  \end{itemize}

  \vspace{0.3em}
  \textbf{Decision rule:}
  \[
  \texttt{decision} = \begin{cases}
    \texttt{STORE}    & u_{\text{stored}} \ge 0.60 \\
    \texttt{DEFERRED} & 0.45 \le u_{\text{stored}} < 0.60 \\
    \texttt{ABSTAIN}  & u_{\text{stored}} < 0.45 \;\wedge\; p_{\text{ground,max}} < 0.20 \\
    \texttt{DISCARD}  & u_{\text{stored}} < 0.45 \;\wedge\; p_{\text{ground,max}} \ge 0.20
  \end{cases}
  \]

  \textbf{Confabulation early-exit (precedence over the composite):}
  \[
    u_{\text{internal}} \ge 0.70 \;\wedge\; p_{\text{ground,max}} \le 0.20 \;\Rightarrow\; \texttt{DISCARD},\ \text{escalate to Tier 3}.
  \]

  \why{The early-exit catches the dangerous quadrant: model is sure but evidence directly contradicts. The composite alone could miss this because $u_{\text{stored}}$ might still be moderate.}

  \why{Two distinct ABSTAIN and DISCARD outcomes, not one, because each needs a different production response. ABSTAIN means no evidence at all and the answer is hidden. DISCARD means evidence exists but signals disagree and the answer is shown with a hedge.}

  \why{Holding the threshold constant across cycles keeps the precision target comparable across cycles; the cycle-boundary recalibration absorbs model drift into the signal-to-probability mapping instead of the threshold.}
\end{frame}

\begin{frame}{Stage 7 -- production response per decision}
  \scriptsize
  The full production envelope (frozen dataclass) exposes six rendered fields plus the raw decision. Confidence is the continuous $u_{\text{stored}} \in [0, 1]$, rendered as a percentage; the semantic tag carries the failure-mode meaning; the ordinal label is an at-a-glance hint; the icon name accompanies the rendering.

  \vspace{0.3em}
  \begin{tabular}{@{}lccccp{4.6cm}@{}}
    \toprule
    Decision & \textbf{show\_answer} & tag & label & icon & user-facing caveat \\
    \midrule
    STORE    & \good & verified     & high      & check   & None (clean confident answer) \\
    DEFERRED & \good & provisional  & moderate  & tilde   & ``I'll reconsider this at my next learning cycle. Please ask again later for an updated answer.'' \\
    DISCARD  & \good & conflicting  & low       & warning & ``Evidence exists but some signals disagree -- independent verification is recommended.'' \\
    ABSTAIN  & \bad  & insufficient & very\_low & cross   & None; answer replaced with ``I don't have enough evidence to answer this reliably.'' \\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \begin{itemize}\scriptsize
    \item \textbf{Continuous confidence is always shown.} The user sees both a semantic tag and the underlying calibrated probability rendered as a percentage (e.g., \textbf{verified $\cdot$ $72\%$}). The four-class tag describes the failure mode; the percentage gives the granular reliability.
    \item \textbf{Only the answer string is hidden on ABSTAIN}; the substituted insufficient-evidence message replaces the answer, the caveat field stays empty.
    \item \textbf{DEFERRED caveat has three deployment-aware forms.} Generic ``next learning cycle'' is the fallback; an explicit ETA (``around 6 am tomorrow'') overrides it; a named cadence (\emph{nightly} / \emph{hourly} / \emph{continuous} / \emph{shortly}) produces deployment-specific user copy. The cycle itself is query-volume-triggered ($N$ queries per cycle), not wall-clock.
    \item \textbf{DISCARD shows the answer with a hedge.} Evidence exists but signals disagree -- the user gets the model's best guess plus the explicit independent-verification ask.
  \end{itemize}

  \why{Four-class semantic tag plus continuous confidence percentage plus deployment-aware hedge text is the load-bearing trust-calibration surface. A system that always shows or always hides is less informative than one that grades trust; a system with no calibrated probability cannot anchor the tag in a number the user can reason about.}

  \altn{A single binary ``answered / refused'' surface (which plain fine-tuning produces) collapses DEFERRED, DISCARD, and ABSTAIN into one bucket and forces the deployer to choose between hiding too much and exposing too much. The four-class schema preserves the calibrated probability and the failure-mode signal separately.}
\end{frame}

\begin{frame}{Cycle loop -- Step 1: LoRA SIL fine-tune}
  \scriptsize
  Fine-tune LoRA adapters on verified episodes drawn from memory.

  \vspace{0.3em}
  \textbf{Primitive:} LoRA rank $r{=}32$, $\alpha{=}64$, all-linear target modules $\{q, k, v, o, \text{gate}, \text{up}, \text{down}\}$. Loss is plain cross-entropy on the verified pool. The LoRA parameter budget acts as the implicit anchor, so no explicit L$_2$ term is used.

  \vspace{0.3em}
  \textbf{Optimisation envelope:}
  \begin{itemize}\scriptsize
    \item \texttt{torch.optim.AdamW} over the $\sim$60 M adapter parameters at $\text{lr}=2\!\cdot\!10^{-4}$ ($10\times$ the backbone-FT learning rate; the adapter is small enough to take a larger step). The bitsandbytes 8-bit AdamW path was explicitly disabled on the LoRA branch (Phase 1d, 2026-05-10): on a $\sim$60 M adapter the $\sim$50\% optimiser-state saving is $\sim$60 MB on a 32 GB envelope, while the 8-bit moment-quantisation noise is relatively larger per-parameter on a small adapter than on a 3 B backbone. The 8-bit path is retained only for the Full-FT fallback
    \item Gradient checkpointing on, AMP \texttt{bfloat16}
    \item Eager-mode SIL block (avoids compile-backward OOM under attention SDPA)
    \item Batch size $4$, gradient accumulation $4$, effective batch $16$; three epochs per cycle
    \item Training pool: $u_{\text{stored}} \ge \tau_{\text{train}} = 0.75 > \tau_{\text{store}} = 0.60$, so the training-pool threshold is strictly stricter than the storage threshold
    \item Pool reweighting: temperature mixing $T{=}2$, three-fold upsampling cap, a per-benchmark floor (DoReMi-style), and a cold-start gold-label fallback for any benchmark whose verified-candidate count drops to zero. A hard per-benchmark share cap of $40\%$ prevents monoculture
  \end{itemize}

  \vspace{0.3em}
  \textbf{Multi-modal retention guard:} measure MMLU, TriviaQA-test, and CommonsenseQA-test on 200 held-out items each before and after fine-tuning. If \alert{any} probe drops below $\rho_{\min}=0.93$ of its pristine value, abort the cycle, restore the previous adapter, and keep the memory pool intact.

  \why{LoRA $r{=}32$, $\alpha{=}64$, all-linear targets: published configuration matching the parametric-capacity-versus-cost frontier observed for 3B-class instruction-tuned models.}

  \why{$\tau_{\text{train}} > \tau_{\text{store}}$: training-pool quality must be stricter than user-facing storage; otherwise borderline-stored entries contaminate the gradient signal.}

  \why{Multi-probe AND-of-OK retention: MCQ-only retention probes leave open-text generation decay invisible. The three probes span bounded-label and open-text together so neither failure mode is silently tolerated.}
\end{frame}

\begin{frame}{Cycle loop -- Steps 2.1--2.5: recalibrate before retroverify}
  \scriptsize
  \begin{enumerate}\scriptsize
    \item \textbf{Step 2.1} -- score the disjoint calibration fold (500 samples per training benchmark, 1500 pooled) under the post-SIL model; write per-sample signals plus EM labels.
    \item \textbf{Step 2.2} -- refit the per-benchmark temperature $T_b$ by ECE-minimisation; refit the per-benchmark safety floor $u_{\text{pre,min}}^b$.
    \item \textbf{Step 2.3} -- refit the per-signal isotonic curves and the per-benchmark Cherian boost weights; the new composite replaces the in-memory one.
    \item \textbf{Step 2.4} -- hot-reload the verifier from the freshly-written cycle composite, no process restart.
    \item \textbf{Step 2.5} -- retroactive re-verification: rescore every stored entry through the fresh verifier; prune entries that fall below $\tau_{\text{retro}}=0.50$, keep and update the rest.
  \end{enumerate}

  \why{Recalibrate then retroverify (not the reverse): if memory were rescored under the stale verifier first, the post-fine-tune signal-distribution shift would over-prune correct entries. The correct order recalibrates to the new distribution and then rescores memory through that mapping.}

  \why{The threshold itself does not drift across cycles: the gate cut on the calibrated output is held constant. Cycle-boundary motion lives entirely in the signal-to-probability mapping.}

  \why{$\tau_{\text{retro}} = 0.50 > \tau_{\text{defer}} = 0.45$: an entry that is borderline-deferred but no longer storage-quality survives the retroactive prune, so the next cycle's deferred-reconsideration pass gets a fair second look at it.}
\end{frame}

\begin{frame}{Cycle loop -- Steps 4 and 5: stream and held-out evaluation}
  \scriptsize
  \textbf{Step 4 -- stream chunk:} 4700 fresh queries flow through the full eight-stage pipeline under cycle-$N$ calibration with storage enabled. The chunk is split 2000 FEVER, 2000 TriviaQA, 700 CommonsenseQA. STORE and DEFER outcomes accumulate in memory and in the deferred buffer.

  \vspace{0.3em}
  \textbf{Step 4b -- snapshot:} per-sample stream-chunk outputs are snapshotted before the held-out eval overwrites them, so the trajectory-relevant storage-decision data is preserved.

  \vspace{0.3em}
  \textbf{Step 5 -- held-out evaluation:} 1500 disjoint samples (300 per benchmark across the five-benchmark panel) with storage disabled. Headline metrics: exact match, F1, composite hallucination metric, composite evaluation score, storage rate, Tier-1 hit rate.

  \vspace{0.4em}
  \textbf{Benchmark panel:}
  \begin{itemize}\scriptsize
    \item Training: FEVER, TriviaQA, CommonsenseQA. The model sees these in stream chunks.
    \item Transfer: TruthfulQA, StrategyQA. The model never sees these in training.
  \end{itemize}

  \why{The training-transfer split matters for the empirical claim: training-panel improvement could be memorisation, but transfer-panel improvement is the load-bearing evidence that the architecture confers cross-distribution benefit.}

  \textbf{Cycle close:} write the LoRA adapter directory ($\sim$50 MB), the retroactive-verification log, the memory store, and the deferred buffer; archive the whole cycle directory to the public host. Cycle $N{+}1$ begins.
\end{frame}

\begin{frame}{Theoretical results -- 5 theorems organised by theme}
  \scriptsize
  Each theme pairs a \alert{conformal worst-case bound} with a Bayes expected-value identity. ``Conformal'' = finite-sample-provable per cycle; ``asymptotic'' = projection.

  \vspace{0.3em}
  \begin{tabular}{@{}p{1.6cm}p{6.0cm}p{5.5cm}@{}}
    \toprule
    \textbf{Theme} & \textbf{Conformal claim} & \textbf{Bayes identity} \\
    \midrule
    Pool purity & Thm 1: $\pi_t \ge 1 - \alpha_{\text{store}}$ & Thm 2: $P_c = p\alpha/(p\alpha + (1{-}p)(1{-}\alpha))$ \\
    Monotonicity & Thm 3: $\sigma_t$ rises with verifier $d$ & Thm 4: $P_c$ rises with base $p_c$ \\
    Convergence & Thm 5: $\mathbb{E}[G_t] \le (1{-}r)^t G_0$ \\
                & under stationarity (Robbins--Monro \cite{robbins1951stochastic}, Polyak \cite{polyak1990new}) & Thm 6: fixed-point $f(p) \to 1$ as $c\to\infty$ \\
    Asymp.\ floor & \alert{Cor.\ tier1-floor (was Thm 7):} $\mathrm{CHM}_1(\infty) \le \max(\alpha_{\text{store}}, 1{-}\pi_{\text{retro}})$ & Thm 7: full-pipeline $H_t \le (1{-}\tau_1)\varepsilon_{\text{arch}} + H_{\text{mem}}(t)$ \\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \textbf{Plus 3 corollaries + 1 remark:}
  \begin{itemize}\scriptsize
    \item Cor.: Tier-1 floor (corollary of convergence theorem)
    \item Cor.: base-model-conditional convergence rate ($c^*(\varepsilon)$)
    \item Cor.: corpus-bounded floor (proxy via top-$k$ retrieval miss subset)
    \item Cor.: self-correction under improving verifier
    \item Remark: joint optimum saturation (interpretive, not directly measurable)
  \end{itemize}

  \vspace{0.3em}
  {\scriptsize Under the fixed-threshold gate, the conformal-form storage error rate in Thm 1 and Cor.\ Tier-1 floor is stated as the empirical STORE-bucket false-positive rate. The inequality structure is preserved; the precondition is per-benchmark verifier balanced accuracy $> 0.5$, and the constant is the measured per-cycle false-positive rate. Per-benchmark generalisation lets the floor be stated separately for each training benchmark, which the pooled form cannot express.}

  \begin{itemize}\scriptsize

\section*{Positioning}

\begin{frame}{Five differentiators vs plain fine-tuning (Appendix G)}
  \footnotesize
  At the same labelling budget, the simpler baseline (B6 in Ch5) is plain fine-tuning. CAEM's load-bearing wins:

  \vspace{0.4em}
  \begin{enumerate}\scriptsize
    \item \textbf{Per-query precision contract.} A fixed threshold on a per-benchmark calibrated probability decides what enters memory; storage-bucket exact-match is tracked per cycle per benchmark. Fine-tuning emits a string with no confidence axis.
    \item \textbf{Memory-direct serving tier.} The Tier-1 hit rate compounds cost savings as the verified pool grows. Fine-tuning still pays a full forward pass per query.
    \item \textbf{Asymmetric rollback.} A failed cycle restores the LoRA adapter but keeps the memory pool. A failed fine-tuning run loses all gradient progress.
    \item \textbf{Verifier as sample-efficiency multiplier.} The cycle-boundary labelled batch validates the storage gate; the gradient signal does not have to carry calibration directly.
    \item \textbf{Inference-time error catching.} The verifier keeps running after fine-tuning saturates against the parametric capacity ceiling. Fine-tuning has nowhere left to go once that ceiling is hit.
  \end{enumerate}
\end{frame}

% =====================================================================
\begin{frame}{What CAEM does that traditional RAG cannot (statelessness)}
  \footnotesize
  \textbf{Traditional RAG:} retrieve passages $\to$ generate $\to$ \alert{discard everything}. Same query in next session $\to$ pay full retrieval+generation cost again, possibly get a different (inconsistent) answer.

  \vspace{0.5em}
  \textbf{CAEM:} retrieve $\to$ generate $\to$ verify $\to$ \emph{store the verified answer}. Future similar queries hit the memory tier in $\sim$50 ms.

  \vspace{0.5em}
  \textbf{Compounded benefits:}
  \begin{enumerate}\small
    \item \textbf{Cost amortisation.} The Tier-3 RAG cost ($\sim$10 s) is paid ONCE; every future Tier-1 hit on the same neighbourhood pays only $\sim$50 ms.
    \item \textbf{Answer consistency.} The same query in different sessions returns the SAME stored answer. Traditional RAG can return inconsistent answers because retrieval and generation both have stochastic components.
    \item \textbf{Domain-specific knowledge accumulation.} A deployment in (say) legal QA accumulates verified legal answers over time -- the system gets specifically better at the queries the user actually asks.
    \item \textbf{Self-improvement via cycle-boundary fine-tune.} Verified episodes feed Step 1 SIL fine-tune. The model learns from its own RAG-mediated correctness. Traditional RAG cannot do this because it has no memory of which retrievals led to correct answers.
    \item \textbf{Quality control on accumulation.} The per-benchmark composite and fixed-threshold gate ensure the memory only admits entries whose calibrated probability clears $0.60$. Storage-bucket precision is tracked per cycle; entries falling below $\tau_{\text{retro}}=0.50$ at the next cycle boundary are retroactively pruned.
  \end{enumerate}
\end{frame}

% =====================================================================
\begin{frame}{What CAEM does NOT solve (honest limitations)}
  \scriptsize
  \begin{itemize}
    \item \textbf{Cold-start hallucinations.} Cycles 0--1 have small memory + low Tier-1 share, so most queries hit Tier 3 with only the verifier as defense. Pool purity benefits emerge only after memory accumulates.
    \item \textbf{Verifier judge ceiling.} MiniCheck NLI itself can mis-classify (the worked example showed $p_{\text{entail}} = 0.06$ on a correct answer). Composite + isotonic mitigate but cannot exceed the judge's ceiling.
    \item \textbf{Cross-encoder rerank cost.} Reranker is the largest single wall-clock component ($\sim$2.4 s/sample) and is NOT cacheable across cycles because the rerank pair includes the model's answer (which changes per cycle).
    \item \textbf{High-confidence rare-fact false negatives.} Architecture trades these for the precision contract -- registered, deliberate, but non-trivial cost.
    \item \textbf{Distribution shift outside the calibration fold.} The fixed-threshold gate provides an empirical precision target rather than a distribution-free coverage guarantee. Queries from a fundamentally different distribution may violate the empirical precision pattern until the next cycle-boundary calibration fold from that new distribution refits the per-benchmark composite. Transfer-benchmark precision is the load-bearing signal here.
    \item \textbf{Compute cost at training time.} Per-cycle cal+retroverify+stream+eval $\approx$ 14 hours on a 5090 with current optimisations. Plain fine-tuning is faster per cycle in pure wall-clock (but weaker on every other axis).
    \item \textbf{Memory pollution risk under bias.} If the cold-start seed contains systematic bias, the verifier can perpetuate it via SIL fine-tune. Mitigation: integrity envelope NFR5 (fold disjointness) + retroactive prune at $\tau_{\text{retro}}$.
  \end{itemize}
\end{frame}
