\section*{Frontmatter}

\begin{frame}{The problem this thesis solves}
  \begin{itemize}
    \item Large language models hallucinate at \textbf{3\%--91\%} across deployment domains
          \tiny\cite{alkaissi2023artificial,10.1145/3649506}\normalsize.
    \item TruthfulQA shows hallucination is \textbf{not solved by scale} — bigger models often hallucinate \emph{more confidently}
          \tiny\cite{lin-etal-2022-truthfulqa}\normalsize.
    \item Existing fixes are \alert{post-hoc detectors}: they label outputs but cannot \emph{change what the system serves}.
  \end{itemize}

  \vspace{0.6em}
  \begin{block}{Thesis claim}
    Treat hallucination as a \textbf{system-level architectural} problem, not a detection problem.
    Pair the model with an \emph{episodic memory}, a \emph{calibrated nine-signal verifier}, a \emph{conformal storage gate}, and a \emph{cycle-boundary self-improvement loop} that learns from its own verified outputs.
  \end{block}

  \vspace{0.4em}
  Five concrete deliverables that plain fine-tuning structurally cannot produce:
  per-query precision contract, memory-direct serving tier, asymmetric rollback,
  verifier-as-sample-efficiency-multiplier, inference-time error catching.
\end{frame}

% =====================================================================
\begin{frame}{CAEM at a glance — two interlocking loops}
  \begin{columns}[t]
    \column{0.55\textwidth}
    \textbf{Per-query path (8 stages, $\sim$10\,s):}
    \begin{enumerate}\setlength\itemsep{0pt}\small
      \item Pre-routing confidence $u_{\text{pre}}$
      \item Encode + episodic-memory lookup
      \item Three-tier router (with safety override)
      \item Tier execution (memory / zero-shot / RAG)
      \item Nine-signal verifier ensemble (4 families)
      \item Composite + conformal gate $\to$ decision
      \item Output (STORE / DEFER / DISCARD / ABSTAIN)
      \item Per-query housekeeping (memory write)
    \end{enumerate}

    \column{0.45\textwidth}
    \textbf{Per-cycle update (every 9000 queries):}
    \begin{itemize}\setlength\itemsep{0pt}\small
      \item SIL fine-tune w/ L2 anchor + retention guard
      \item Cal-fold scoring under post-SIL model
      \item Re-fit $T$, isotonic curves, conformal $\tau$
      \item Hot-reload verifier (no process restart)
      \item Retroactive re-verify all stored entries
      \item Stream chunk (9000 queries) + eval-fold (3500)
    \end{itemize}
  \end{columns}

  \vspace{0.4em}
  {\footnotesize Backbone: Qwen-2.5-3B-Instruct. Verifier judge: MiniCheck (Flan-T5-Large). Encoder: SBERT all-mpnet-base-v2. Retriever: 21M-passage DPR over Wikipedia. Reranker: BAAI/bge-reranker-v2-m3.}
\end{frame}

\section*{Per-stage deep dive}

\begin{frame}{Stage 1 — pre-routing confidence $u_{\text{pre}}$}
  Two cheap signals from one short forward pass (`caem/memory/entry.py:209`):
  \begin{itemize}\small
    \item $u_{\text{token}}$ — geometric mean of per-token log-probabilities of the draft answer
    \item $c_{\text{conv}}$ — encoder-layer convergence ratio (variance of early layers / variance of late layers)
  \end{itemize}

  \[
    u_{\text{pre}} \;=\; \sigma\!\left(\frac{\operatorname{logit}\bigl(0.60\,u_{\text{token}} + 0.40\,(1{+}c_{\text{conv}})^{-1}\bigr)}{T}\right)
  \]
  $T$ is re-fit at every cycle by ECE-minimisation on the 1500-sample held-out cal-fold (\code{config.temperature\_scalar}; updated by Step 2.2).

  \vspace{0.3em}
  \why{The 0.60 / 0.40 split favours token-level confidence (cheap, well-calibrated by SFT) over the convergence ratio (noisier signal). Both reach into model internals so neither requires an extra forward pass.}

  \altn{Single-signal $u_{\text{pre}} = u_{\text{token}}$ alone misses the encoder-layer instability that confidently-wrong outputs exhibit. Including $c_{\text{conv}}$ adds $<5$\% latency and catches a class of failures token-prob alone cannot.}

  \vspace{0.3em}
  \textbf{Used by:} (i) router safety override (Stage 3); (ii) Stage-6 ABSTAIN gate; (iii) Step 2.2 T re-fit input.
\end{frame}

% =====================================================================
\begin{frame}{Stages 2--3 — episodic memory and the three-tier router}
  \footnotesize
  \textbf{Memory:} FAISS index over SBERT (all-mpnet-base-v2) embeddings of past verified $(q, a)$ pairs. Top-1 search returns similarity $s$ and the neighbour's stored quality $u_{\text{stored,nbr}}$.

  \vspace{0.3em}
  \textbf{Combined dispatch score:}
  \[
    \mathcal{S} \;=\; \lambda \cdot s \;+\; (1{-}\lambda)\cdot u_{\text{stored,nbr}}, \quad \lambda = 0.70 \quad\text{(\code{config.routing\_lambda})}.
  \]

  \textbf{Dispatch rules in fixed evaluation order (\code{caem/routing/router.py:149}):}
  \begin{enumerate}
    \item \alert{Safety override}: if $u_{\text{pre}} < 0.60 \Rightarrow$ Tier 3 unconditionally.
    \item Tier 1 (memory-direct) if $\mathcal{S} \ge 0.90$ — serve neighbour, NO generation, NO verification.
    \item Tier 2 (zero-shot) if similarity $> 0.75$ AND not Tier 1 — generate without retrieval.
    \item Tier 3 (RAG) — fallback, retrieve top-K passages + grounded generation.
  \end{enumerate}

  \why{Three tiers, not two, because the binary memory-vs-generate decision over-burdens a single threshold. Tier 2 absorbs the ``in-distribution but novel question'' band where the model can answer zero-shot but no neighbour is close enough to trust.}

  \why{Safety override is OR, not AND: the OR-condition expresses that model readiness AND memory quality are \emph{both} mandatory --- low confidence wins regardless of how good the neighbour is.}

  \altn{No memory tier (RAG-only baseline B2): pays full retrieval+generation cost on every query, no memorisation benefit. Plain fine-tuning baseline B6: pays full forward-pass on every query and offers no per-query confidence axis.}
\end{frame}

% =====================================================================
\begin{frame}{Stage 4 — tier execution: why three handlers}
  \footnotesize
  \begin{tabular}{@{}p{2cm}p{4cm}p{6cm}@{}}
    \toprule
    \textbf{Tier} & \textbf{What it does} & \textbf{Cost / when chosen} \\
    \midrule
    Tier 1 (memory) & Returns neighbour's stored answer; verifier skipped & $\sim$50 ms / query;
                    chosen when memory has a match committed under $u_{\text{stored}} \ge \tau_{\text{store}}$ and similarity $\ge 0.97$-ish (combined $\ge 0.90$). \\
    Tier 2 (zero-shot) & Single forward pass without retrieval; verifier runs & $\sim$3--5 s / query;
                    chosen for in-distribution queries where the model's own knowledge suffices. \\
    Tier 3 (RAG) & Retrieve from 21M passages, rerank with BAAI/bge-reranker, generate grounded; verifier runs & $\sim$10 s / query;
                    architectural floor when nothing else has earned it. \\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \textbf{Cost--quality compounding:}
  \begin{itemize}\small
    \item Early cycles: Tier 1 share $\approx 0$\%, Tier 3 share $> 90$\%.
    \item Late cycles: Tier 1 share grows as the verified pool fills; mean per-query latency drops accordingly.
    \item This is a load-bearing differentiator vs plain fine-tuning, which still pays a full forward pass per query forever.
  \end{itemize}

  \why{Verifier is deliberately SKIPPED on Tier 1: the entry was committed under the conformal gate at storage time, so re-verifying every recall would (a) double the compute and (b) be circular --- the verifier would just confirm what the prior gate already established. This is the per-query precision contract acting like a paid receipt.}
\end{frame}

% =====================================================================
\begin{frame}{Stage 5 -- Family A: internal calibration}
  Three signals from the model itself. White-box, zero extra forward passes.

  \vspace{0.3em}
  \begin{tabular}{@{}lp{4.5cm}p{5cm}@{}}
    \toprule
    Signal & Definition & Purpose \\
    \midrule
    $u_{\text{token}}$ & geometric mean per-token log-prob of draft & calibrated by SFT, cheap, baseline confidence \\
    $u_{\text{dropout}}$ & variance over $K=5$ MC-Dropout passes (rate $0.10$) & catches answers stable under deterministic decode but fragile under noise \\
    $u_{\text{internal}}$ & $\tfrac{1}{2}\,u_{\text{token}} + \tfrac{1}{2}(1 - u_{\text{dropout}})$ & combined internal stability signal \\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \why{$K=5$ is the published Bayesian-dropout balance \tiny\cite{NEURIPS2017_84ddfb34}\normalsize: enough to estimate predictive variance, few enough to keep the per-query cost under one extra forward pass equivalent.}

  \why{Combination via $u_{\text{internal}}$ is the input to the Stage-6 confabulation early-exit. Linear combination chosen over multiplication so the signals stay in $[0,1]$ and the boost layer can re-weight them on log-odds without sign-flipping artefacts.}

  \altn{Just $u_{\text{token}}$ alone: published research on \cite{kadavath2022language} confirms token-prob misses the over-confident ``I'm sure'' failure mode. Adding $u_{\text{dropout}}$ catches that.}
\end{frame}

% =====================================================================
\begin{frame}{Stage 5 -- Family B: sample-set agreement}
  \footnotesize
  Resample $M{=}3$ chains at $T{=}0.7$ and measure inter-chain agreement.

  \vspace{0.4em}
  \begin{tabular}{@{}lp{5cm}p{5cm}@{}}
    \toprule
    Signal & Definition & Closes which failure mode \\
    \midrule
    $s_{\text{avg}}$ & Mean unique-pair cosine over $M{=}3$ chains in SBERT space & ``self-consistency'': the same model produces the same answer when re-rolled \tiny\cite{wang2023selfconsistency}\normalsize \\
    $p_{\text{entail}}$ & Mean MiniCheck NLI entailment of each chain $\to$ served answer & ``the served answer is well-formed but is NOT what the chains support'' \\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \why{$M=3$ is the wall-clock floor for self-consistency to discriminate stable from unstable answers --- $M=1$ gives zero variance, $M=10$ adds little discriminative power per Wang et al.}

  \why{Why $T=0.7$ and not greedy: greedy ($T=0$) collapses all chains to the same string, $s_{\text{avg}} = 1$ trivially. $T = 1.0$ is too noisy; $T = 0.7$ is the empirical sweet spot from the published self-consistency line.}

  \alert{Note:} the original 10th candidate $h_{\text{norm}}$ (normalised semantic entropy at $K=10$, $T=1.0$) was empirically retired --- the cycle-0 calibration audit fitted a boost weight of \alert{$-5 \times 10^{-4}$}, statistically zero. It cost $\sim$4\,s/sample for no information gain. Receipt: \code{cycle\_0/composite\_calibration.json}.
\end{frame}

% =====================================================================
\begin{frame}{Stage 5 -- Family C: external grounding (and the rerank choice)}
  \footnotesize
  Three signals over passages retrieved from a 21M-passage Wikipedia DPR index, reranked by BAAI/bge-reranker-v2-m3:

  \vspace{0.3em}
  \begin{tabular}{@{}lp{6cm}p{3.5cm}@{}}
    \toprule
    Signal & Pair scored under MiniCheck NLI & Catches \\
    \midrule
    $p_{\text{ground,max}}$ & strongest of top-3 reranked passages $\to$ answer & ``one passage clearly supports'' \\
    $p_{\text{ground,mean}}$ & mean over all top-3 $\to$ answer & ``broad support across passages'' \\
    $p_{\text{ground,atomic}}$ & length-gated atomic-fact decomp $\to$ each top-3 & ``answer claims something not in any passage'' \cite{min-etal-2023-factscore} \\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \alert{Important architectural detail:} the reranker scores pairs of \alert{$(query + answer, \text{passage})$}, not just $(query, \text{passage})$ (\code{verifier.py:1822}).

  \vspace{0.2em}
  \why{Concatenating the answer to the query forces the reranker to find passages that could substantiate the \emph{specific claim} the model just produced --- the load-bearing input to $p_{\text{ground,*}}$. If we ranked by topical relevance to the question only, the top passages might be on-topic but say nothing about the answer.}

  \why{This is also why the rerank is NOT cacheable across cycles: the served answer changes after every SIL fine-tune, so the rerank pair is different and the top-3 changes. The cost is paid every cycle by design.}
\end{frame}

% =====================================================================
\begin{frame}{Stage 5 -- Family D + composite math}
  \footnotesize
  \textbf{Family D (one signal):} $q_{\text{a,rel}}$ -- independent cross-encoder score of $(\text{question}, \text{served\_answer})$. The verifier's cross-encoder is FROZEN; the generator does NOT influence it.

  \vspace{0.4em}
  \why{This single signal carries the strongest weight ($w = 0.500$) in the boost layer. The reason: it's the only signal that catches the ``off-topic but grounded'' failure mode --- model confidently answers a related-but-different question and retrieval finds passages supporting the wrong question. Internal/grounding signals miss this; q-a relevance catches it.}

  \vspace{0.5em}
  \textbf{Stage 5b -- calibrated probability composite (\code{cal\_prob\_composite.py:280}):}
  \begin{enumerate}\small
    \item For each signal $j$: $P_j = \mathrm{iso}_j(\text{signal}_j) \in [0,1]$. \quad (per-signal isotonic regression, fit on cycle-0 cal-fold)
    \item Convert to log-odds: $L_j = \log\bigl(P_j/(1{-}P_j)\bigr)$.
    \item L$_2$-regularised logistic boost \tiny\cite{cherian2024conformal}\normalsize: $\quad\operatorname{logit}(u_{\text{stored}}) = b + \sum_j w_j L_j$.
    \item Sigmoid back: $u_{\text{stored}} = \sigma(\cdot) \in (0,1)$.
  \end{enumerate}

  \why{Per-signal isotonic before the boost: each signal has its own monotone but non-linear calibration curve --- e.g.\ MiniCheck entailment isn't linear in $P(\text{correct})$. Isotonic absorbs that without parametric assumptions.}

  \why{Why L$_2$ not L$_1$: every signal in the registered set is expected to carry non-zero discriminative signal under the registered hypotheses; L$_2$ shrinks weights smoothly without setting any to zero, preserving the architectural commitment to nine-signal coverage. \altn{L$_1$ would zero some weights and turn the boost layer into a feature-selection step, which is what the cycle-0 audit performs separately and explicitly (h_norm retirement).}}

  \why{$C = 0.01$ (strong regulariser) chosen by the 25-variant pre-registered Step 7.0.3 sweep over $C \in \{0.01, 0.10, 1.00\}$. Locked because empirical evaluation-fold ID precision dominated.}
\end{frame}

\begin{frame}{Stage 6 -- four-outcome decision tree}
  \footnotesize
  \textbf{Conformal split-CP gate} fitted on the cycle-0 cal-fold (\code{outputs/cycle\_0/conformal\_gate.json}):

  \begin{itemize}\small
    \item $\alpha_{\text{store}} = 0.05 \Rightarrow$ precision target $0.95$ \quad $\to \tau_{\text{store}} = 0.6676$ \quad ($n_{\text{store}} = 56$, $\pi_{\text{store}} = 0.9643$)
    \item $\alpha_{\text{defer}} = 0.40 \Rightarrow$ precision target $0.60$ \quad $\to \tau_{\text{defer}} = 0.5207$ \quad ($n_{\text{defer}} = 191$, $\pi_{\text{defer}} = 0.6021$)
  \end{itemize}

  \vspace{0.3em}
  \textbf{Decision rule} (verifier.py:65--70):
  \[
  \texttt{decision} = \begin{cases}
    \texttt{STORE}    & u_{\text{stored}} \ge \tau_{\text{store}} \\
    \texttt{DEFERRED} & \tau_{\text{defer}} \le u_{\text{stored}} < \tau_{\text{store}} \\
    \texttt{ABSTAIN}  & u_{\text{stored}} < \tau_{\text{defer}} \;\wedge\; p_{\text{ground,max}} < \phi_{\text{pg}} \\
    \texttt{DISCARD}  & u_{\text{stored}} < \tau_{\text{defer}} \;\wedge\; p_{\text{ground,max}} \ge \phi_{\text{pg}}
  \end{cases}
  \]
  with $\phi_{\text{pg}} = 0.20$ (the abstention floor; \emph{distinct} from the safety floor $0.60$ on $u_{\text{pre}}$).

  \vspace{0.2em}
  \textbf{Confabulation early-exit} (precedence over the composite, \code{verifier.py:54}):
  \[
    u_{\text{internal}} \ge 0.70 \;\wedge\; p_{\text{ground,max}} \le 0.20 \;\Rightarrow\; \texttt{DISCARD},\ \text{escalate to Tier 3}.
  \]

  \why{The early-exit catches the dangerous quadrant ``model is sure but evidence directly contradicts'' --- this is observationally distinct from low-confidence answers and the composite alone could miss it because $u_{\text{stored}}$ might still be moderate.}

  \why{Two distinct ABSTAIN/DISCARD outcomes (not one) because they need different production responses: ABSTAIN means ``no evidence at all, hide the answer''; DISCARD means ``some evidence but signals disagree, show with a hedge''.}
\end{frame}

  \begin{itemize}\small
    \item Confidence is always shown; only the answer text is hidden (ABSTAIN).

\begin{frame}{Cycle-boundary loop -- Step 1: SIL fine-tune}
  \footnotesize
  Fine-tune the generator on verified episodes from memory + a general-domain mix (\code{caem/training/self\_improvement.py}).

  \vspace{0.3em}
  \textbf{Loss function:}
  \[
    \mathcal{L} \;=\; \mathcal{L}_{\mathrm{CE}}(\theta) \;+\; \lambda_{\text{anc}}\,\|\theta - \theta_{t-1}\|_{2}^{2}.
  \]
  $\theta_{t-1}$ kept on CPU and streamed per-parameter via PCIe (frees $\sim$6 GB GPU; OOM-mitigation patch).

  \vspace{0.3em}
  \textbf{Optimisation envelope:}
  \begin{itemize}\small
    \item AdamW8bit (bitsandbytes) with gradient checkpointing and AMP bf16
    \item \code{torch.compiler.set\_stance("force\_eager")} for the SIL block (avoids the SDPA-compiled-backward OOM)
    \item batch\_size $= 4$, grad\_accum\_steps $= 4 \Rightarrow$ effective batch $= 16$
    \item epochs\_per\_cycle $= 3$
  \end{itemize}

  \vspace{0.3em}
  \textbf{Retention guard:} measure MMLU on 200 held-out items pre + post fine-tune. If
  \[
    \rho = \frac{\text{MMLU}_{\text{post}}}{\text{MMLU}_{\text{pristine}}} \;<\; \rho_{\min} = 0.93,
  \]
  \alert{abort the cycle}: restore $\theta_{t-1}$, \alert{KEEP the accumulated memory pool}. This is asymmetric rollback.

  \why{L$_2$ anchor is a first-order approximation to elastic weight consolidation \cite{doi:10.1073/pnas.1611835114}. Full EWC requires the per-weight Fisher diagonal, which we cannot compute on a 3.086\,B-parameter generator within the cycle budget. The uniform-Fisher approximation bounds drift cheaply.}

  \why{$\rho_{\min}=0.93$ rather than $1.00$: a strict lower bound would abort too aggressively (small noise drops can cross 1.00), 0.93 is the published continual-learning practical floor and matches MMLU's measurement noise.}
\end{frame}

% =====================================================================
\begin{frame}{Cycle-boundary loop -- Steps 2.1--2.5: recalibration before retroverify}
  \footnotesize
  \begin{enumerate}\small
    \item \textbf{Step 2.1} -- score the same disjoint 1500-sample cal-fold under the post-SIL model. Cache $u_{\text{pre}}$ + 9 signals + EM per sample to \code{cycle\_N/calibration/\{bench\}\_cycle\{N\}.json}.
    \item \textbf{Step 2.2} -- re-fit $T$ by ECE-min on (cached $u_{\text{pre}}$, em). \alert{Reads the cache} written by 2.1 instead of re-running the pipeline (saves $\sim$3.5 h per cycle; recent optimisation).
    \item \textbf{Step 2.3} -- re-fit per-signal isotonic curves AND conformal $\tau$ on the post-SIL cal-fold. EMA-smooth: $\tau_{\text{new}} = 0.7\,\tau_{\text{prev}} + 0.3\,\tau_{\text{fit}}$ (\code{adaptive\_thresholds\_ema\_alpha = 0.7}).
    \item \textbf{Step 2.4} -- hot-reload the verifier's calibration JSONs in the running process (no restart).
    \item \textbf{Step 2.5} -- retroactive re-verification: re-score every stored episode through the freshly-recalibrated verifier; prune below $\tau_{\text{retro}} = 0.50$, keep + upgrade upward otherwise.
  \end{enumerate}

  \why{Recalibrate THEN retroverify (not the reverse): if we retroverified under the stale verifier, we'd re-score memory through cycle-$N{-}1$'s thresholds and over-prune EM-correct entries because the post-SIL signal distribution has shifted. Correct order: first recalibrate to the new distribution, then re-score memory.}

  \why{EMA factor $0.7$ keeps thresholds slow-moving --- a higher factor would freeze the thresholds (calibration drift accumulates), a lower factor would chase single-cycle noise (storage rate oscillates). 0.7 is the empirical compromise.}

  \why{$\tau_{\text{retro}} = 0.50$ < $\tau_{\text{defer}} = 0.52$ deliberately: an entry borderline-deferred but no longer storage-quality is still kept for the deferred-reconsider pass at the next cycle, rather than categorically deleted.}
\end{frame}

% =====================================================================
\begin{frame}{Cycle-boundary loop -- Steps 4 + 5: stream + eval}
  \footnotesize
  \textbf{Step 4 -- cycle-$N$ stream chunk:} 9000 fresh queries (3000 each from the training panel: FEVER, TriviaQA, NQ) flow through the full Stage 1--7 pipeline under cycle-$N$ calibration. STORE/DEFER outcomes accumulate into memory and the deferred buffer respectively.

  \vspace{0.3em}
  \textbf{Step 5 -- cycle-$N$ eval-fold:} 3500 disjoint samples ($500 \times 7$ benchmarks) measured per cycle. Writes per-sample JSON to \code{eval/\{bench\}\_cycle\{N\}.json}. Headline metrics (EM, F1, CHM, CES, storage rate, Tier-1 hit rate) come from these.

  \vspace{0.4em}
  \textbf{Benchmark panel structure (\code{run\_phase1a.sh}):}
  \begin{itemize}\small
    \item Training panel (3): FEVER, TriviaQA, NQ-Open --- the model SEES these in stream chunks.
    \item Held-out transfer panel (4): TruthfulQA, StrategyQA, ARC-Challenge, ASQA --- the model NEVER sees these in training. Test of generalisation.
  \end{itemize}

  \why{Splitting the panel matters for the empirical claim: training-panel improvement could be memorisation, but transfer-panel improvement is the load-bearing evidence that the architecture confers cross-distribution benefit.}

  \why{$N=3000$ training queries per cycle balances cycle-budget vs information per cycle. Smaller would dilute the signal of any per-cycle verifier improvement; larger would cost more wall-clock without proportional benefit.}

  \textbf{Cycle close:} save \code{cycle\_N/model.pt} ($\sim$6.17 GB), gdrive offload, append a row to \code{experiment\_summary.csv}. Cycle $N{+}1$ begins.
\end{frame}

\section*{Theory and evidence}

\begin{frame}{Four theoretical guarantees (each paired with empirical receipt)}
  \footnotesize
  \begin{enumerate}\setlength\itemsep{0.4em}
    \item \textbf{Pool purity floor.} Under upward-only retroactive update: \alert{$\pi_t \ge 1 - \alpha_{\text{store}}$ for all $t$}.
          \emph{Receipt:} per-cycle pool purity should not drop below $\min(0.95, \pi_{\text{retro}})$ at any cycle. Reading: \code{outputs/full\_run/pool\_purity\_per\_cycle.csv}.

    \item \textbf{Monotonicity.} Storage rate AND pool purity are monotonic in the verifier's discrimination $d$ (Cohen's $d$ on cal-fold) AND in the base-model accuracy $p_+$.
          \emph{Receipt:} $d_t$ vs $\sigma_t$ vs $\pi_t$ trajectory across the 10 cycles in Ch5 figure.

    \item \textbf{Geometric envelope.} The precision-asymptote gap satisfies $|\pi_t - \pi_\infty| \le A\,r^{t}$ with contraction rate $r = \mathcal{O}\!\bigl(\sigma\,\Phi(d/\sqrt{2})\bigr)$.
          \emph{Receipt:} exponential-decay fit to $G_t = |\pi_t - \pi_\infty|$ across cycles; Phase 4 produces \code{purity\_curve\_fit.json}.

    \item \textbf{Asymptotic Tier-1 hallucination floor.} The memory-direct tier's hallucination rate is bounded:
          \[
            H_\infty \;\le\; \alpha_{\text{store}} + (1{-}\alpha_{\text{store}})\,\varepsilon_{\text{arch}}.
          \]
          The floor is set by the conformal level $\alpha_{\text{store}}$, NOT by deployment scale. \alert{This is the load-bearing theoretical claim} — scale alone cannot beat $\alpha_{\text{store}}$.
  \end{enumerate}
\end{frame}

\section*{Positioning, limitations, conclusion}

\begin{frame}{Why CAEM beats plain exact-match-labelled fine-tuning}
  At the same labelling budget, the simpler baseline (B6 in Ch5) is plain fine-tuning. CAEM's load-bearing wins:

  \vspace{0.4em}
  \begin{enumerate}\small
    \item \textbf{Per-query precision contract.} Conformal-bounded $1{-}\alpha_{\text{store}} = 0.95$ at every served STORE; fine-tuning emits a string with no confidence axis.
    \item \textbf{Memory-direct serving tier.} Tier-1 hit-rate compounds cost savings as the verified pool grows; fine-tuning still pays a full forward pass per query.
    \item \textbf{Asymmetric rollback.} A failed cycle restores weights but \emph{keeps memory}; fine-tuning's failed run loses all gradient progress.
    \item \textbf{Verifier as sample-efficiency multiplier.} The cycle-boundary labelled batch validates a precision contract; the gradient signal does not have to carry it directly.
    \item \textbf{Inference-time error catching.} Verifier keeps running after the fine-tune saturates against the parametric capacity ceiling. Fine-tuning has nowhere left to go once that ceiling is hit.
  \end{enumerate}
\end{frame}

% =====================================================================
\begin{frame}{What CAEM does NOT solve (honest limitations)}
  \begin{itemize}\small
    \item \textbf{Cold-start hallucinations.} Cycles 0--1 have a small memory and a low Tier-1 share, so most queries hit Tier 3 with only the verifier as defense. Pool purity benefits only emerge once the memory accumulates.
    \item \textbf{Verifier judge ceiling.} MiniCheck NLI itself can mis-classify (the worked example showed $p_{\text{entail}} = 0.06$ on a correct answer). The composite + isotonic mitigate this but cannot exceed the judge's inherent ceiling.
    \item \textbf{Cross-encoder rerank cost.} The reranker is the largest single wall-clock component ($\sim$2.4 s/sample) and is not cacheable across cycles because the rerank pair includes the model's answer.
    \item \textbf{High-confidence rare-fact false negatives.} The architecture trades these for the precision contract --- a registered, deliberate, but non-trivial cost.
    \item \textbf{Distribution shift outside the calibration fold.} Conformal guarantees hold under exchangeability; queries from a fundamentally different distribution may violate the precision lower bound until the next cal-fold drawn from the new distribution restores calibration.
    \item \textbf{Compute cost at training time.} Per-cycle calibration + retroverify + stream + eval $\approx$ 14 hours on a 5090 with current optimisations. Plain fine-tuning is faster per cycle in pure wall-clock.
  \end{itemize}
\end{frame}
