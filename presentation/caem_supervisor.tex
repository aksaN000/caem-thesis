\section*{Problem and gap}

\begin{frame}{The problem}
  \begin{itemize}\small
    \item Large language models hallucinate at \textbf{3\%--91\%} across deployment domains
          \tiny\cite{alkaissi2023artificial,10.1145/3649506}\normalsize.
    \item TruthfulQA shows hallucination is \textbf{not solved by scale} -- bigger models often hallucinate \emph{more confidently} \tiny\cite{lin-etal-2022-truthfulqa}\normalsize.
    \item Existing fixes are \alert{post-hoc detectors}: they label outputs but cannot \emph{change what the system serves}.
  \end{itemize}

  \vspace{0.5em}
  \begin{block}{Thesis claim}
    Treat hallucination as a \textbf{system-level architectural} problem, not a detection problem.
    Pair the model with an \emph{episodic memory}, a \emph{calibrated nine-signal verifier}, a \emph{conformal storage gate}, and a \emph{cycle-boundary self-improvement loop} that learns from its own verified outputs.
  \end{block}
\end{frame}

% =====================================================================
\begin{frame}{The three structural shortcomings of stateless inference}
  Standard inference suffers \alert{three reinforcing shortcomings} (Ch1 \S stateless-gap):

  \vspace{0.5em}
  \begin{enumerate}\small
    \item \textbf{Statelessness.} Every query processed independently. \emph{No representation of past successful answers retained}. The system wastes computation on questions it has effectively already solved, and identical questions can produce inconsistent answers from one session to the next.
    \item \textbf{No calibrated confidence.} The model emits a string without a per-query reliability signal. The user cannot tell trustworthy outputs apart from confident-but-wrong ones.
    \item \textbf{No self-improvement mechanism.} External verifications (whether by humans or other systems) decay into one-shot interventions. The model learns nothing from its own history of right and wrong answers.
  \end{enumerate}

  \vspace{0.5em}
  \begin{block}{What CAEM gives back}
    Memory closes (1). Calibrated $u_{\text{stored}}$ + four-outcome decision closes (2). Cycle-boundary SIL fine-tune closes (3). Each axis reinforces the others -- without memory the model cannot reuse verified knowledge; without calibration the system cannot recognise when memory is missing; without self-improvement, knowledge accumulated stays static.
  \end{block}
\end{frame}

% =====================================================================
\begin{frame}{CAEM at a glance -- two interlocking loops}
  \begin{columns}[t]
    \column{0.55\textwidth}
    \textbf{Per-query path (8 stages, $\sim$10\,s):}
    \begin{enumerate}\setlength\itemsep{0pt}\scriptsize
      \item Pre-routing confidence $u_{\text{pre}}$
      \item Encode + episodic-memory lookup
      \item Three-tier router (with safety override)
      \item Tier execution (memory / zero-shot / RAG)
      \item Nine-signal verifier ensemble
      \item Composite + conformal gate $\to$ decision
      \item Output (STORE / DEFER / DISCARD / ABSTAIN)
      \item Per-query housekeeping (memory write)
    \end{enumerate}

    \column{0.45\textwidth}
    \textbf{Per-cycle update (every 9000 queries):}
    \begin{itemize}\setlength\itemsep{0pt}\scriptsize
      \item SIL fine-tune w/ L2 anchor + retention guard
      \item Cal-fold scoring under post-SIL model
      \item Re-fit $T$, isotonic curves, conformal $\tau$
      \item Hot-reload verifier
      \item Retroactive re-verify all stored entries
      \item Stream chunk (9000) + eval-fold (3500)
    \end{itemize}
  \end{columns}

  \vspace{0.4em}
  {\scriptsize Backbone: Qwen-2.5-3B-Instruct. Verifier judge: MiniCheck (Flan-T5-Large). Encoder: SBERT all-mpnet-base-v2. Retriever: 21M-passage DPR over Wikipedia. Reranker: BAAI/bge-reranker-v2-m3.}
\end{frame}

\section*{Goals, requirements, hypotheses, commitments}

\begin{frame}{Specific objectives 1--4 (Ch1 \S specific-objectives)}
  \scriptsize
  \begin{enumerate}
    \item \textbf{Verified episodic memory with a four-outcome decision.} Admission rule finer than a binary threshold; every generated answer enters one of \{STORE, DEFERRED, DISCARD, ABSTAIN\} via the calibrated probability composite + the confabulation early-exit. Storage is similarity-indexed with novelty filter and value-based pruning combining recency, retrieval frequency, and stored-answer quality.
    \item \textbf{Pre-generation confidence and three-tier routing with safety override.} $u_{\text{pre}}$ from encoder-only forward pass, temperature-scaled on a disjoint fold, used \emph{independently} from memory similarity. Router combines query-to-memory similarity with stored-answer quality into a single dispatch score over \{memory-direct, zero-shot, RAG\}. Independent safety override forces RAG whenever $u_{\text{pre}}<0.60$ -- grounded retrieval is available before generation begins.
    \item \textbf{Nine-signal post-generation verifier with calibrated probability composite.} Four families: internal calibration, sample-set agreement, external grounding (3 rules), Q-A relevance. Each signal goes through per-signal isotonic, then through L$_2$-regularised logistic boost; replaces fixed weighted sums used in earlier verifier ensembles.
    \item \textbf{Conformal storage gate with two operating points.} Composite $\to$ decision via split-conformal at pre-registered $\alpha_{\text{store}}=0.05$ (precision target 0.95) and $\alpha_{\text{defer}}=0.40$ (target 0.60). Distribution-free precision guarantee transfers approximately to the eval fold under exchangeability.
  \end{enumerate}
\end{frame}

% =====================================================================
\begin{frame}{Specific objectives 5--7 (Ch1 \S specific-objectives)}
  \scriptsize
  \begin{enumerate}\setcounter{enumi}{4}
    \item \textbf{Retroactive review and deferred reconsideration.} Verifier internal-calibration signals read the generator's hidden state, so stored composites drift after every fine-tune. Retroactive re-verification at every cycle boundary re-scores stored entries under the updated model and prunes entries below $\tau_{\text{retro}}=0.50$. Deferred-buffer pass promotes entries that crossed $\tau_{\text{store}}$ and discards those past TTL.
    \item \textbf{Constrained self-improvement with retention guard.} Training pool curated from stored episodes with $u_{\text{stored}} \ge \tau_{\text{train}} = 0.75$ (\emph{strictly above} the storage threshold $\tau_{\text{store}}=0.6676$). Fine-tune under L$_2$ anchor toward base parameters (first-order EWC). Held-out probe before+after; abort with weight rollback if retention $<\rho_{\min}=0.93$. Memory survives rollback (asymmetric rollback).
    \item \textbf{Empirical validation against external baselines and ablations.} 7-benchmark panel: 3 training (FEVER, TQA, NQ) + 4 transfer (TruthfulQA, StrategyQA, ARC-Challenge, ASQA), with MMLU as held-out retention probe. Composite hallucination metric, per-tier behaviour, retention ratio, calibration trajectory cycle by cycle. 7 external baselines under matched protocol with paired McNemar + BCa bootstrap under Holm correction; pre-registered ablations.
  \end{enumerate}

  \vspace{0.3em}
  {\scriptsize Each objective is a separately-checkable empirical deliverable. The empirical chapter reports the receipt for every one.}
\end{frame}

\begin{frame}{Six design commitments (Ch3 \S constraints)}
  \scriptsize
  \begin{enumerate}
    \item \textbf{Single-backbone commitment.} Generator FIXED across every cycle and every tier. \emph{All reported gains are attributable to memory + calibration + the loop, not to a backbone swap.}
    \item \textbf{Pooled-calibration commitment.} Composite + conformal thresholds fitted ONCE on a calibration fold pooled across training benchmarks; applied verbatim at every scoring site. \emph{No per-benchmark calibration leakage.}
    \item \textbf{Local-execution commitment.} No external model API in the inference path. Verifier ensemble + embedding encoder + passage index + generator all run from open weights on a single rented GPU. \emph{Deployable without third-party dependencies.}
    \item \textbf{Bounded-cycle commitment.} SIL loop capped at 10 cycles; early stop only on the registered equilibrium diagnostic. \emph{Ensures the empirical claim is not propped up by an arbitrary horizon.}
    \item \textbf{Matched-protocol commitment.} Every external baseline evaluated under same samples, prompts, scoring rules. \emph{The comparison panel is not confounded by re-implementation choices.}
    \item \textbf{Single-seed reproducibility.} Random-state surface capped at one recorded seed; per-cycle artefact catalogue + pre-Step-7 snapshot on a public dataset host. \emph{Trajectory regenerable end-to-end by a third party.}
  \end{enumerate}
\end{frame}

\section*{Per-stage deep dive}

\begin{frame}{Stage 1 -- pre-routing confidence $u_{\text{pre}}$}
  \footnotesize
  Two cheap signals from one short forward pass (\code{caem/memory/entry.py:209}):
  \begin{itemize}\small
    \item $u_{\text{token}}$ -- geometric mean of per-token log-probabilities of the draft answer
    \item $c_{\text{conv}}$ -- encoder-layer convergence ratio (variance early / variance late layers)
  \end{itemize}
  \[
    u_{\text{pre}} = \sigma\!\left(\frac{\operatorname{logit}\bigl(0.60\,u_{\text{token}} + 0.40\,(1{+}c_{\text{conv}})^{-1}\bigr)}{T}\right)
  \]
  $T$ refit at every cycle by ECE-min on 1500-sample cal-fold (Step 2.2).

  \vspace{0.2em}
  \why{0.60/0.40 split favours token-level confidence (cheap, well-calibrated by SFT) over the noisier convergence ratio. Both reach into model internals -- no extra forward pass.}

  \altn{Single-signal $u_{\text{pre}} = u_{\text{token}}$: misses the encoder-layer instability that confidently-wrong outputs exhibit. Adding $c_{\text{conv}}$ catches a class token-prob alone cannot.}
\end{frame}

% =====================================================================
\begin{frame}{Stages 2--3 -- memory and the three-tier router}
  \footnotesize
  \textbf{Memory:} FAISS index over SBERT embeddings of past verified $(q,a)$ pairs. Top-1 returns similarity $s$ and neighbour's $u_{\text{stored,nbr}}$.

  \vspace{0.2em}
  \textbf{Combined dispatch score:} $\mathcal{S} = 0.70\,s + 0.30\,u_{\text{stored,nbr}}$ (\code{routing\_lambda=0.70}).

  \vspace{0.2em}
  \textbf{Dispatch rules in fixed order} (\code{router.py:149}):
  \begin{enumerate}\small
    \item Safety override: if $u_{\text{pre}} < 0.60 \Rightarrow$ Tier 3 unconditionally.
    \item Tier 1 (memory-direct) if $\mathcal{S} \ge 0.90$ -- serve neighbour, NO generation, NO verifier.
    \item Tier 2 (zero-shot) if similarity $> 0.75$ AND not Tier 1 -- generate without retrieval.
    \item Tier 3 (RAG) -- fallback floor.
  \end{enumerate}

  \why{Three tiers because a binary memory-vs-generate decision over-burdens a single threshold. Tier 2 absorbs the ``in-distribution but novel question'' band where the model knows zero-shot but no neighbour is close enough.}

  \why{Safety override is OR not AND: model readiness AND memory quality are \emph{both} mandatory -- low confidence wins regardless of how good the neighbour is.}
\end{frame}

\begin{frame}{Stage 4 -- tier execution: why three handlers}
  \scriptsize
  \begin{tabular}{@{}p{1.7cm}p{4.2cm}p{6.5cm}@{}}
    \toprule
    \textbf{Tier} & \textbf{What it does} & \textbf{Cost / when chosen} \\
    \midrule
    Tier 1 (memory) & Returns neighbour's stored answer; verifier skipped & $\sim$50 ms / query;
                    chosen when memory has match committed under $u_{\text{stored}}\ge\tau_{\text{store}}$ (combined $\ge 0.90$). \\
    Tier 2 (zero-shot) & Single forward pass, no retrieval; verifier runs & $\sim$3--5 s / query;
                    in-distribution queries where the model's own knowledge suffices. \\
    Tier 3 (RAG) & Retrieve from 21M passages, rerank, generate grounded; verifier runs & $\sim$10 s / query;
                    architectural floor when nothing else has earned it. \\
    \bottomrule
  \end{tabular}

  \vspace{0.5em}
  \textbf{Cost--quality compounding} (load-bearing for the ``cost reduction'' claim H5):
  \begin{itemize}\scriptsize
    \item Early cycles: Tier 1 share $\approx 0$\%, Tier 3 share $> 90$\% -- pay full RAG cost.
    \item Late cycles: Tier 1 share grows as the verified pool fills.
    \item Mean per-query latency drops $\propto$ Tier-1 share. Plain fine-tuning still pays a full forward pass forever -- this is one of CAEM's load-bearing differentiators.
  \end{itemize}

  \why{Verifier SKIPPED on Tier 1: the entry was committed under the conformal gate at storage time, so re-verifying every recall would (a) double the compute and (b) be circular. The per-query precision contract acts like a paid receipt.}
\end{frame}

\begin{frame}{Stage 5 -- Family D + composite math}
  \scriptsize
  \textbf{Family D (one signal):} $q_{\text{a,rel}}$ -- independent cross-encoder score of $(\text{question}, \text{served\_answer})$. Verifier's cross-encoder is FROZEN.

  \why{This single signal carries the largest weight ($w = 0.500$). It catches the ``off-topic but grounded'' failure mode -- model confidently answers a related-but-different question, retrieval finds passages supporting the wrong question. Internal/grounding signals miss this.}

  \vspace{0.4em}
  \textbf{Stage 5b -- composite (\code{cal\_prob\_composite.py:280}):}
  \begin{enumerate}\scriptsize
    \item For each signal $j$: $P_j = \mathrm{iso}_j(\text{signal}_j) \in [0,1]$ (isotonic, fit on Cycle-0 cal-fold).
    \item $L_j = \log(P_j/(1{-}P_j))$.
    \item L$_2$ logistic boost (Cherian 2024): $\operatorname{logit}(u_{\text{stored}}) = b + \sum_j w_j L_j$.
    \item Sigmoid back: $u_{\text{stored}} = \sigma(\cdot) \in (0,1)$.
  \end{enumerate}

  \why{Per-signal isotonic before boost: each signal has its own monotone but non-linear calibration curve. Isotonic absorbs that without parametric assumptions.}

  \why{L$_2$ not L$_1$: every signal carries non-zero discriminative signal under the registered hypotheses; L$_2$ shrinks smoothly without setting weights to zero (preserves architectural commitment to nine-signal coverage).}

  \why{$C = 0.01$ chosen by 25-variant Step 7.0.3 sweep over $C \in \{0.01, 0.10, 1.00\}$, locked by best ID precision.}
\end{frame}

\begin{frame}{Stage 6 -- four-outcome decision tree}
  \scriptsize
  \textbf{Conformal split-CP gate} (\code{outputs/cycle\_0/conformal\_gate.json}):
  \begin{itemize}\scriptsize
    \item $\alpha_{\text{store}} = 0.05 \Rightarrow$ precision target $0.95$ \quad $\to \tau_{\text{store}} = 0.6676$ \quad ($n_{\text{store}} = 56$, $\pi_{\text{store}} = 0.9643$)
    \item $\alpha_{\text{defer}} = 0.40 \Rightarrow$ precision target $0.60$ \quad $\to \tau_{\text{defer}} = 0.5207$
  \end{itemize}

  \vspace{0.3em}
  \textbf{Decision rule} (\code{verifier.py:65--70}):
  \[
  \texttt{decision} = \begin{cases}
    \texttt{STORE}    & u_{\text{stored}} \ge \tau_{\text{store}} \\
    \texttt{DEFERRED} & \tau_{\text{defer}} \le u_{\text{stored}} < \tau_{\text{store}} \\
    \texttt{ABSTAIN}  & u_{\text{stored}} < \tau_{\text{defer}} \;\wedge\; p_{\text{ground,max}} < 0.20 \\
    \texttt{DISCARD}  & u_{\text{stored}} < \tau_{\text{defer}} \;\wedge\; p_{\text{ground,max}} \ge 0.20
  \end{cases}
  \]

  \textbf{Confabulation early-exit (precedence over composite):}
  \[
    u_{\text{internal}} \ge 0.70 \;\wedge\; p_{\text{ground,max}} \le 0.20 \;\Rightarrow\; \texttt{DISCARD},\ \text{escalate to Tier 3}.
  \]

  \why{The early-exit catches the dangerous quadrant: ``model is sure but evidence directly contradicts.'' Composite alone could miss this because $u_{\text{stored}}$ might still be moderate.}

  \why{Two distinct ABSTAIN/DISCARD outcomes (not one) because each needs a different production response. ABSTAIN: ``no evidence at all, hide answer.'' DISCARD: ``evidence exists, signals disagree, show with hedge.''}
\end{frame}

\begin{frame}{Stage 7 -- production response per decision}
  \footnotesize
  User-facing output rules (\code{caem/production/confidence.py}):

  \vspace{0.3em}
  \begin{tabular}{@{}lcccp{4.2cm}@{}}
    \toprule
    Decision & \textbf{show\_answer} & tag & label & user-facing caveat \\
    \midrule
    STORE    & \good & verified     & high     & --- (clean confident answer) \\
    DEFERRED & \good & provisional  & moderate & ``I'll reconsider this at my\\&&&& next learning cycle.'' \\
    DISCARD  & \good & conflicting  & low      & ``Evidence exists but signals\\&&&& disagree -- verify independently.'' \\
    ABSTAIN  & \bad & insufficient  & very\_low & ``I don't have enough evidence\\&&&& to answer this reliably.'' \\
    \bottomrule
  \end{tabular}

  \vspace{0.4em}
  \begin{itemize}\scriptsize
    \item Confidence is always shown; only the answer text is hidden (ABSTAIN).
    \item DEFERRED + DISCARD show the answer with a hedge -- user is informed the system is uncertain and why.
    \item This matters for trust calibration: a system that always shows or always hides is less informative than one that grades trust.
  \end{itemize}

  \why{The four-class output schema lets the deployer build trust-aware UX. Plain fine-tuning has no such axis -- it just emits a string.}
\end{frame}

\begin{frame}{Cycle loop -- Step 1: SIL fine-tune}
  \scriptsize
  Fine-tune the generator on verified episodes from memory + general-domain mix (\code{caem/training/self\_improvement.py}).

  \vspace{0.3em}
  \textbf{Loss:} $\mathcal{L} = \mathcal{L}_{\mathrm{CE}}(\theta) + \lambda_{\text{anc}}\,\|\theta - \theta_{t-1}\|_2^2$. \quad
  $\theta_{t-1}$ on CPU, streamed per-parameter via PCIe (frees $\sim$6 GB GPU; OOM-mitigation patch).

  \vspace{0.3em}
  \textbf{Optimisation envelope:}
  \begin{itemize}\scriptsize
    \item AdamW8bit (bitsandbytes) + gradient checkpointing + AMP bf16
    \item \code{torch.compiler.set\_stance("force\_eager")} for SIL block (avoids SDPA-compile-backward OOM)
    \item batch\_size $= 4$, grad\_accum\_steps $= 4 \Rightarrow$ effective batch $= 16$; epochs\_per\_cycle $= 3$
    \item Training pool: $u_{\text{stored}} \ge \tau_{\text{train}} = 0.75 > \tau_{\text{store}} = 0.6676$ -- training threshold STRICTER than storage
  \end{itemize}

  \vspace{0.3em}
  \textbf{Retention guard:} measure MMLU on 200 held-out items pre+post fine-tune. If $\rho = \mathrm{MMLU}_{\text{post}}/\mathrm{MMLU}_{\text{pristine}} < \rho_{\min} = 0.93$, \alert{abort cycle}, restore $\theta_{t-1}$, KEEP memory pool.

  \why{L$_2$ anchor is first-order EWC (Kirkpatrick 2017). Full EWC requires per-weight Fisher diagonal -- can't compute on 3.086\,B params within cycle budget. Uniform-Fisher approximation bounds drift cheaply.}

  \why{$\rho_{\min} = 0.93$ not $1.00$: strict bound aborts on small noise; 0.93 is the published continual-learning practical floor matching MMLU's measurement noise.}

  \why{$\tau_{\text{train}} > \tau_{\text{store}}$: training pool quality MUST be stricter than user-facing storage. Otherwise borderline-stored entries contaminate the gradient signal.}
\end{frame}

% =====================================================================
\begin{frame}{Cycle loop -- Steps 2.1--2.5: recalibrate before retroverify}
  \scriptsize
  \begin{enumerate}\scriptsize
    \item \textbf{Step 2.1} -- score the same disjoint 1500-sample cal-fold under post-SIL model. Cache $u_{\text{pre}}$ + 9 signals + EM per sample to \code{cycle\_N/calibration/}.
    \item \textbf{Step 2.2} -- re-fit $T$ by ECE-min on (cached $u_{\text{pre}}$, em). \alert{Reads cache} from 2.1 instead of re-running pipeline (saves $\sim$3.5 h/cycle).
    \item \textbf{Step 2.3} -- re-fit per-signal isotonic + conformal $\tau$ on post-SIL fold. EMA-smooth: $\tau_{\text{new}} = 0.7\,\tau_{\text{prev}} + 0.3\,\tau_{\text{fit}}$.
    \item \textbf{Step 2.4} -- hot-reload verifier in running process (no restart).
    \item \textbf{Step 2.5} -- retroactive re-verification: re-score every stored entry through fresh verifier; prune below $\tau_{\text{retro}} = 0.50$, keep + upgrade upward otherwise.
  \end{enumerate}

  \why{Recalibrate THEN retroverify (NOT reverse): if we retroverified under the stale verifier, we'd re-score memory through cycle-$N{-}1$'s thresholds and over-prune EM-correct entries because the post-SIL signal distribution shifted. Correct order: first recalibrate to new distribution, then re-score memory.}

  \why{EMA factor $0.7$: keeps thresholds slow-moving. Higher $\Rightarrow$ frozen (calibration drift accumulates). Lower $\Rightarrow$ chases single-cycle noise. 0.7 is the empirical compromise.}

  \why{$\tau_{\text{retro}} = 0.50 < \tau_{\text{defer}} = 0.52$ deliberately: an entry borderline-deferred but no longer storage-quality is still kept for the deferred-reconsider pass at next cycle, rather than categorically deleted.}
\end{frame}

\begin{frame}{Cycle loop -- Steps 4 + 5: stream + eval}
  \scriptsize
  \textbf{Step 4 -- cycle-$N$ stream chunk:} 9000 fresh queries (3000 each from training panel: FEVER, TQA, NQ) flow through full Stage 1--7 under cycle-$N$ calibration. STORE/DEFER outcomes accumulate in memory + deferred buffer.

  \vspace{0.3em}
  \textbf{Step 5 -- cycle-$N$ eval-fold:} 3500 disjoint samples ($500 \times 7$ benchmarks) per cycle. Per-sample JSON to \code{eval/\{bench\}\_cycle\{N\}.json}. Headline metrics: EM, F1, CHM, CES, storage rate, Tier-1 hit rate.

  \vspace{0.4em}
  \textbf{Benchmark panel structure:}
  \begin{itemize}\scriptsize
    \item Training panel (3): FEVER, TriviaQA, NQ-Open -- model SEES these in stream chunks.
    \item Held-out transfer panel (4): TruthfulQA, StrategyQA, ARC-Challenge, ASQA -- model NEVER sees these in training.
  \end{itemize}

  \why{Splitting matters for the empirical claim: training-panel improvement could be memorisation, but transfer-panel improvement is the load-bearing evidence that the architecture confers cross-distribution benefit.}

  \why{$N=3000$ training queries per cycle: balance cycle-budget vs information per cycle. Smaller dilutes the verifier-improvement signal; larger costs more wall-clock without proportional benefit.}

  \textbf{Cycle close:} save \code{cycle\_N/model.pt} ($\sim$6.17 GB), gdrive offload, append row to \code{experiment\_summary.csv}. Cycle $N{+}1$ begins.
\end{frame}

\section*{Theory and evidence}

\begin{frame}{Four theoretical guarantees (each with empirical receipt)}
  \scriptsize
  \begin{enumerate}\setlength\itemsep{0.4em}
    \item \textbf{Theorem 1 -- Pool purity floor.} Under upward-only retroactive update: \alert{$\pi_t \ge 1 - \alpha_{\text{store}}$ for all $t$}.
          \emph{Receipt:} per-cycle pool purity $\ge \min(0.95, \pi_{\text{retro}})$. \emph{Hypothesis:} H1, H2.

    \item \textbf{Theorem 2 -- Monotonicity.} Storage rate AND pool purity both rise with verifier discrimination $d$ (Cohen's $d$ on cal-fold) AND base-model accuracy $p_+$.
          \emph{Receipt:} $d_t$ vs $\sigma_t$ vs $\pi_t$ trajectory across 10 cycles in Ch5 figure.

    \item \textbf{Theorem 3 -- Geometric envelope.} $|\pi_t - \pi_\infty| \le A\,r^{t}$ with contraction rate $r = \mathcal{O}\!\bigl(\sigma\,\Phi(d/\sqrt{2})\bigr)$.
          \emph{Receipt:} exponential-decay fit to gap $G_t = |\pi_t - \pi_\infty|$ across cycles. \emph{Hypothesis:} H3.

    \item \textbf{Theorem 4 -- Asymptotic Tier-1 hallucination floor.}
          \[
            H_\infty \;\le\; \alpha_{\text{store}} + (1{-}\alpha_{\text{store}})\,\varepsilon_{\text{arch}}.
          \]
          The floor is set by $\alpha_{\text{store}}$, NOT by deployment scale. \alert{This is the load-bearing theoretical claim} -- scale alone cannot beat $\alpha_{\text{store}}$. \emph{Hypothesis:} H4.
  \end{enumerate}
\end{frame}

\section*{Positioning}

\begin{frame}{Five differentiators vs plain fine-tuning (Appendix G)}
  \footnotesize
  At the same labelling budget, the simpler baseline (B6 in Ch5) is plain fine-tuning. CAEM's load-bearing wins:

  \vspace{0.4em}
  \begin{enumerate}\scriptsize
    \item \textbf{Per-query precision contract.} Conformal-bounded $1{-}\alpha_{\text{store}} = 0.95$ at every served STORE; fine-tuning emits a string with no confidence axis.
    \item \textbf{Memory-direct serving tier.} Tier-1 hit-rate compounds cost savings as the verified pool grows; fine-tuning still pays a full forward pass per query.
    \item \textbf{Asymmetric rollback.} Failed cycle restores weights but \emph{keeps memory}; fine-tuning's failed run loses all gradient progress.
    \item \textbf{Verifier as sample-efficiency multiplier.} Cycle-boundary labelled batch validates a precision contract; gradient signal does not have to carry it directly.
    \item \textbf{Inference-time error catching.} Verifier keeps running after fine-tune saturates against the parametric capacity ceiling. Fine-tuning has nowhere left to go once that ceiling is hit.
  \end{enumerate}
\end{frame}

% =====================================================================
\begin{frame}{What CAEM does that traditional RAG cannot (statelessness)}
  \footnotesize
  \textbf{Traditional RAG:} retrieve passages $\to$ generate $\to$ \alert{discard everything}. Same query in next session $\to$ pay full retrieval+generation cost again, possibly get a different (inconsistent) answer.

  \vspace{0.5em}
  \textbf{CAEM:} retrieve $\to$ generate $\to$ verify $\to$ \emph{store the verified answer}. Future similar queries hit the memory tier in $\sim$50 ms.

  \vspace{0.5em}
  \textbf{Compounded benefits:}
  \begin{enumerate}\small
    \item \textbf{Cost amortisation.} The Tier-3 RAG cost ($\sim$10 s) is paid ONCE; every future Tier-1 hit on the same neighbourhood pays only $\sim$50 ms.
    \item \textbf{Answer consistency.} The same query in different sessions returns the SAME stored answer. Traditional RAG can return inconsistent answers because retrieval and generation both have stochastic components.
    \item \textbf{Domain-specific knowledge accumulation.} A deployment in (say) legal QA accumulates verified legal answers over time -- the system gets specifically better at the queries the user actually asks.
    \item \textbf{Self-improvement via cycle-boundary fine-tune.} Verified episodes feed Step 1 SIL fine-tune. The model learns from its own RAG-mediated correctness. Traditional RAG cannot do this because it has no memory of which retrievals led to correct answers.
    \item \textbf{Quality control on accumulation.} Conformal gate ensures the memory only fills with provably high-precision entries ($\ge 95\%$). Verified pool grows monotonically in quality.
  \end{enumerate}
\end{frame}

% =====================================================================
\begin{frame}{What CAEM does NOT solve (honest limitations)}
  \scriptsize
  \begin{itemize}
    \item \textbf{Cold-start hallucinations.} Cycles 0--1 have small memory + low Tier-1 share, so most queries hit Tier 3 with only the verifier as defense. Pool purity benefits emerge only after memory accumulates.
    \item \textbf{Verifier judge ceiling.} MiniCheck NLI itself can mis-classify (the worked example showed $p_{\text{entail}} = 0.06$ on a correct answer). Composite + isotonic mitigate but cannot exceed the judge's ceiling.
    \item \textbf{Cross-encoder rerank cost.} Reranker is the largest single wall-clock component ($\sim$2.4 s/sample) and is NOT cacheable across cycles because the rerank pair includes the model's answer (which changes per cycle).
    \item \textbf{High-confidence rare-fact false negatives.} Architecture trades these for the precision contract -- registered, deliberate, but non-trivial cost.
    \item \textbf{Distribution shift outside the calibration fold.} Conformal guarantees hold under exchangeability; queries from a fundamentally different distribution may violate the precision lower bound until the next cal-fold from the new distribution restores calibration.
    \item \textbf{Compute cost at training time.} Per-cycle cal+retroverify+stream+eval $\approx$ 14 hours on a 5090 with current optimisations. Plain fine-tuning is faster per cycle in pure wall-clock (but weaker on every other axis).
    \item \textbf{Memory pollution risk under bias.} If the cold-start seed contains systematic bias, the verifier can perpetuate it via SIL fine-tune. Mitigation: integrity envelope NFR5 (fold disjointness) + retroactive prune at $\tau_{\text{retro}}$.
  \end{itemize}
\end{frame}
